--- TABLE: turmas ---

codigo | varchar(20)
ano_id | int(11)

numero_turma | tinyint(4)
sala_principal | varchar(20)
vagas | int(11)
ativa | tinyint(1)
data_criacao | timestamp

--- TABLE: disciplinas ---
id | int(11)
codigo | varchar(10)
nome | varchar(100)
ano_id | int(11)
carga_horaria | int(11)
credito | decimal(3,1)
descricao | text
ativa | tinyint(1)
data_criacao | timestamp

--- TABLE: horarios ---
id | int(11)
turma_id | int(11)

dia_semana | enum('Segunda','Terça','Quarta','Quinta','Sexta','Sábado')
